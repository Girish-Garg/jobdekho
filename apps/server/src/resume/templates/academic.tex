% Academic: formal serif, suited to a research or teaching record. Uses
% LaTeX's default Computer Modern rather than an extra font package, since a
% CV in this register is conventionally set in the plain LaTeX face.
%
% Same ATS-safe rules as the other templates: no tables, no multi-column
% flow, standard \section headings, plain selectable text, URLs shown as
% text. Black on white throughout - hierarchy comes from size and weight.
%
% Placeholders (filled by resume/render.js, never edited by hand):
%   NAME, HEADLINE, CONTACT, BODY - see resume/render.js for the fill() step
\documentclass[11pt]{article}

% cmap before fontenc: without it, Computer Modern's "fi"/"fl" ligatures
% compile fine but extract as missing text ("notifications" reads back as
% "notications"), which is exactly the kind of copy-paste and ATS-parsing
% failure this template exists to avoid.
\usepackage{cmap}
\usepackage[margin=1in]{geometry}
\usepackage[T1]{fontenc}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage[hidelinks]{hyperref}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\raggedbottom
\raggedright

% The rule sits under the heading: titlesec's \titlerule after a point of
% space. It used to be drawn after pulling back up 6pt, which ran it through
% the heading's letters so every section read as struck out.
\titleformat{\section}{\normalsize\scshape}{}{0pt}{}[\vspace{1pt}\titlerule]
\titlespacing*{\section}{0pt}{12pt}{5pt}

\newenvironment{resItems}{%
  \begin{itemize}[leftmargin=1.2em, topsep=2pt, itemsep=1.5pt, parsep=0pt]
}{%
  \end{itemize}
}

% Structural macros. Only these definitions differ between templates: the
% content generated by resume/sections.js calls the same macro names no
% matter which template renders it.
% A break belongs before content that exists, never after content that may
% not. "{\itshape #2}\\" with an empty #2 is the "There's no line here to end"
% error that every profile without a headline used to compile to. The test
% is e-TeX's \detokenize, so an argument starting with a control sequence
% (an escaped & from a company name) is still read as non-empty.
\newcommand{\resHeaderLine}[3]{\if\relax\detokenize{#3}\relax\else\\[#1]{#2 #3}\fi}
\newcommand{\resHeader}[3]{%
  \begin{center}
    {\Large #1}%
    \resHeaderLine{4pt}{\itshape}{#2}%
    \resHeaderLine{4pt}{\small}{#3}%
  \end{center}
  \vspace{2pt}\hrule
}
\newcommand{\resSection}[1]{\section*{#1}}
\newcommand{\resEntryHeader}[3]{%
  \noindent\textbf{#1}\hfill{#3}\\
  \noindent\textit{#2}\par
}
\newcommand{\resEntryHeaderPlain}[2]{%
  \noindent\textbf{#1}\hfill{#2}\par
}
\newcommand{\resMetaLine}[1]{{\small #1}\par}
\newcommand{\resEntrySpace}{\vspace{7pt}}
\newcommand{\resSkillRow}[2]{\noindent\textit{#1:}\ #2\par\vspace{2pt}}

\begin{document}

\resHeader{@@NAME@@}{@@HEADLINE@@}{@@CONTACT@@}
\vspace{4pt}

@@BODY@@

\end{document}
