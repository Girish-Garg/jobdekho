% Compact: tighter modern sans, built to fit more onto one page.
%
% Same ATS-safe rules as classic.tex: no tables, no multi-column flow,
% standard \section headings, plain selectable text, contact URLs shown as
% text and an entry's own links by name (see classic.tex for why). The
% document stays black on white - hierarchy comes from size and weight, not
% colour, same as the rest of this application.
%
% Placeholders (filled by resume/render.js, never edited by hand):
%   NAME, HEADLINE, CONTACT, BODY - see resume/render.js for the fill() step
\documentclass[10pt]{article}

% cmap before fontenc: keeps "fi"/"fl" ligatures copy-pasting and
% ATS-parsing back out as themselves instead of vanishing from the text.
\usepackage{cmap}
\usepackage[margin=0.55in]{geometry}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage[hidelinks]{hyperref}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\raggedbottom
\raggedright

% The rule sits under the heading: titlesec's \titlerule after a point of
% space. It used to be drawn after pulling back up 6pt, which ran it through
% the heading's letters so every section read as struck out.
% \MakeUppercase takes the title as its argument in the before-code slot;
% in the format slot it grabbed titlesec's own row macro instead.
\titleformat{\section}{\normalsize\bfseries}{}{0pt}{\MakeUppercase}[\vspace{1pt}\titlerule]
\titlespacing*{\section}{0pt}{7pt}{4pt}

\newenvironment{resItems}{%
  \begin{itemize}[leftmargin=1em, topsep=1pt, itemsep=0.5pt, parsep=0pt]
}{%
  \end{itemize}
}

% Structural macros. Only these definitions differ between templates: the
% content generated by resume/sections.js calls the same macro names no
% matter which template renders it.
% A break belongs before content that exists, never after content that may
% not. "{\normalsize #2}\\" with an empty #2 is the "There's no line here to
% end" error that every profile without a headline used to compile to. The
% test is e-TeX's \detokenize, so an argument that starts with a control
% sequence (an escaped & from a company name) is still read as non-empty.
\newcommand{\resHeaderLine}[3]{\if\relax\detokenize{#3}\relax\else\\[#1]{#2 #3}\fi}
\newcommand{\resHeader}[3]{%
  {\Large\bfseries #1}%
  \resHeaderLine{1pt}{\small}{#2}%
  \resHeaderLine{1pt}{\footnotesize}{#3}%
  \par\vspace{2pt}
  \hrule
}
\newcommand{\resSection}[1]{\section*{#1}}
\newcommand{\resEntryHeader}[3]{%
  \noindent\textbf{#1}\hfill\textit{\footnotesize #3}\\
  \noindent{\footnotesize #2}\par
}
\newcommand{\resEntryHeaderPlain}[2]{%
  \noindent\textbf{#1}\hfill\textit{\footnotesize #2}\par
}
\newcommand{\resMetaLine}[1]{{\footnotesize #1}\par}
\newcommand{\resEntrySpace}{\vspace{3pt}}
\newcommand{\resSkillRow}[2]{\noindent{\small\textbf{#1:}\ #2}\par\vspace{1pt}}

\end{document}
