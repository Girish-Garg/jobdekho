% Compact: tighter modern sans, built to fit more onto one page.
%
% Same ATS-safe rules as classic.tex: no tables, no multi-column flow,
% standard \section headings, plain selectable text, URLs shown as text. The
% document stays black on white - hierarchy comes from size and weight, not
% colour, same as the rest of this application.
%
% Placeholders (filled by resume/render.js, never edited by hand):
%   NAME, HEADLINE, CONTACT, BODY - see resume/render.js for the fill() step
\documentclass[10pt]{article}

% cmap before fontenc: keeps "fi"/"fl" ligatures copy-pasting and
% ATS-parsing back out as themselves instead of vanishing from the text.
\usepackage{cmap}
\usepackage[margin=0.55in]{geometry}
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\renewcommand{\familydefault}{\sfdefault}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage[hidelinks]{hyperref}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\raggedbottom
\raggedright

\titleformat{\section}{\normalsize\bfseries\MakeUppercase}{}{0pt}{}[{\vspace{-5pt}\hrule\vspace{5pt}}]
\titlespacing*{\section}{0pt}{7pt}{2pt}

\newenvironment{resItems}{%
  \begin{itemize}[leftmargin=1em, topsep=1pt, itemsep=0.5pt, parsep=0pt]
}{%
  \end{itemize}
}

% Structural macros. Only these definitions differ between templates: the
% content generated by resume/sections.js calls the same macro names no
% matter which template renders it.
\newcommand{\resHeader}[3]{%
  {\Large\bfseries #1}\\[1pt]
  {\small #2}\\[1pt]
  {\footnotesize #3}\\[2pt]
  \hrule
}
\newcommand{\resSection}[1]{\section*{#1}}
\newcommand{\resEntryHeader}[3]{%
  \noindent\textbf{#1}\hfill\textit{\footnotesize #3}\\
  \noindent{\footnotesize #2}\par
}
\newcommand{\resEntryHeaderPlain}[2]{%
  \noindent\textbf{#1}\hfill\textit{\footnotesize #2}\par
}
\newcommand{\resMetaLine}[1]{{\footnotesize #1}\par}
\newcommand{\resEntrySpace}{\vspace{3pt}}
\newcommand{\resSkillRow}[2]{\noindent{\small\textbf{#1:}\ #2}\\[1pt]}

\end{document}
